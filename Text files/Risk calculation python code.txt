# Losses are defined as negative simple returns.
# VaR and ES are reported as positive loss magnitudes.
def tail_metrics(returns, confidence):
    r = pd.Series(returns, dtype="float64").dropna().to_numpy()

    if len(r) < 20:
        raise ValueError("Too few observations for this historical estimate")

    losses = -r
    var = float(np.quantile(losses, confidence, method="linear"))
    tail_losses = losses[losses >= var]
    es = float(tail_losses.mean())

    return {
        "VaR": var,
        "ES_CVaR": es,
        "Tail_Observations": int(len(tail_losses)),
    }

def return_metrics(returns):
    r = pd.Series(returns, dtype="float64").dropna()

    if len(r) < 2:
        raise ValueError("At least two observations are required")

    wealth = (1 + r).cumprod()
    peak = wealth.cummax()
    drawdown = wealth / peak - 1

    downside = np.minimum(r.to_numpy(), 0.0)

    return {
        "Observations": int(len(r)),
        "Mean_Daily_Return": float(r.mean()),
        "Daily_Return_StdDev": float(r.std(ddof=1)),
        "Annualized_Volatility": float(r.std(ddof=1) * np.sqrt(252)),
        "Compounded_Return": float(wealth.iloc[-1] - 1),
        "Maximum_Drawdown": float(drawdown.min()),
        "Negative_Return_Days": int((r < 0).sum()),
        "Annualized_Downside_Deviation": float(
            np.sqrt(np.mean(downside ** 2)) * np.sqrt(252)
        ),
    }

# Each period/strategy is evaluated separately.

}
